function copyToClipboard(text, label) {
  var chinese = document.documentElement.lang === 'zh-CN';
  var displayLabel = chinese ? (label === 'Email' ? '邮箱' : '微信号') : label;
  function showToast(message) {
    var toast = document.getElementById('copy-toast');
    toast.textContent = message;
    toast.classList.add('show');
    setTimeout(function() {
      toast.classList.remove('show');
    }, 2000);
  }
  if (!navigator.clipboard) {
    showToast(chinese ? '请手动复制已显示的联系方式' : 'Please copy the displayed contact manually.');
    return;
  }
  navigator.clipboard.writeText(text).then(function() {
    showToast(chinese ? '✓ ' + displayLabel + '已复制' : '✓ ' + label + ' copied!');
  }).catch(function() {
    showToast(chinese ? '复制失败，请手动复制' : 'Unable to copy. Please copy manually.');
  });
}

function revealAndCopyContact(button, text, label) {
  button.textContent = text + ' ';
  var icon = document.createElement('i');
  icon.className = 'fas fa-copy';
  icon.setAttribute('aria-hidden', 'true');
  button.appendChild(icon);
  button.classList.add('copyable');
  copyToClipboard(text, label);
}
</script>
